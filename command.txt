config-app-deployment.yaml
===============
apiVersion: apps/v1
kind: Deployment
metadata:
  name: config-app
spec:
  replicas: 2
  selector:
    matchLabels:
      app: config-app
  template:
    metadata:
      labels:
        app: config-app
    spec:
      containers:
      - name: web
        image: nginx:alpine
        command: ["/bin/sh", "-c"]
        args:
        - |
          cat > /usr/share/nginx/html/index.html <<EOF
          <html>
          <body>
            <h1>Configuration Management Demo</h1>
            <p>Mode: ${APP_MODE}</p>
            <p>${WELCOME_MESSAGE}</p>
            <p>DB User configured: ${DB_USERNAME}</p>
            <p>DB Password: [MASKED]</p>
          </body>
          </html>
          EOF
          nginx -g 'daemon off;'
        env:
        - name: APP_MODE
          valueFrom:
            configMapKeyRef:
              name: app-config
              key: APP_MODE
        - name: WELCOME_MESSAGE
          valueFrom:
            configMapKeyRef:
              name: app-config
              key: WELCOME_MESSAGE
        - name: DB_USERNAME
          valueFrom:
            secretKeyRef:
              name: app-secret
              key: DB_USERNAME
        - name: DB_PASSWORD
          valueFrom:
            secretKeyRef:
              name: app-secret
              key: DB_PASSWORD
        ports:
        - containerPort: 80
        volumeMounts:
        - name: secret-volume
          mountPath: /etc/app-secret
          readOnly: true
        - name: config-volume
          mountPath: /etc/app-config
          readOnly: true
      volumes:
      - name: secret-volume
        secret:
          secretName: app-secret
      - name: config-volume
        configMap:
          name: app-config
-------------
kubectl apply -f config-app-deployment.yaml
kubectl get pods -l app=config-app
